Save the following code as `results/recheck_core_overlap.py`, then run it from


The equivalent saved snippet was executed against this frozen subset with exit
code 0. It confirms the exact counts and fractions listed in
[C1](../../../experiments/core_rq/CLAIMS.md#c1).

```python
import csv, hashlib, io, json
from collections import Counter
from fractions import Fraction as F
from math import comb
from pathlib import Path
root = Path('experiments/core_rq/overlap_pilot_v1')
manifest = json.loads((root / 'manifest.json').read_bytes())
expected_hash = '593d362a8a5bdcd8dc0366e9d1cb05238c6a07b8cc0c75419691511c53e8e17d'
assert type(manifest['schema_version']) is int and manifest['schema_version'] == 1
assert manifest['configuration']['config_hash'] == expected_hash
tables = {}
for name in ('instances.csv', 'raw_results.csv'):
    data = (root / name).read_bytes()
    assert hashlib.sha256(data).hexdigest() == manifest['outputs'][name]['sha256']
    assert len(data) == manifest['outputs'][name]['bytes']
    tables[name] = list(csv.DictReader(io.StringIO(data.decode('utf-8'))))
instances, runs = tables['instances.csv'], tables['raw_results.csv']
assert len(instances) == 60 and len(runs) == 120
by_id = {x['instance_id']: x for x in instances}
units = {(x['case_id'], int(x['repetition'])): x for x in instances}
assert len(by_id) == len(units) == 60
assert set(units) == {(c, r) for c in ('overlap', 'overlap_control') for r in range(30)}
assert all(x['config_hash'] == expected_hash for x in instances + runs)
lookup = {(x['instance_id'], x['algorithm_id']): x for x in runs}
assert len(lookup) == len({x['run_id'] for x in runs}) == 120
assert set(lookup) == {(i, a) for i in by_id for a in ('greedy', 'exact_reference')}
for run in runs:
    assert all(run[k] == by_id[run['instance_id']][k] for k in ('case_id', 'repetition', 'seed'))
    assert not run['algorithm_seed'] and json.loads(run['algorithm_metadata'])['termination'] == 'completed'
counts, gaps = Counter(), []
for repetition in range(30):
    pair = [units[c, repetition] for c in ('overlap', 'overlap_control')]
    assert pair[0]['seed'] == pair[1]['seed'] and all(not x['coupling_seed'] for x in pair)
    failures, differences = [], []
    for instance in pair:
        g, e = [lookup[instance['instance_id'], a] for a in ('greedy', 'exact_reference')]
        assert g['algorithm'] == 'greedy' and g['status'] == 'feasible'
        assert e['algorithm'] == 'brute_force' and e['status'] == 'optimal'
        assert json.loads(e['algorithm_options']) == {'max_set_count': 16, 'time_limit_seconds': None}
        G, O = int(g['coverage']), int(e['coverage'])
        assert 0 <= G <= O and O > 0 and int(g['optimum']) == int(e['optimum']) == O
        failures.append(int(G < O)); differences.append(F(O - G, O))
    counts[tuple(failures)] += 1; gaps.append(differences)
assert [counts[k] for k in ((0, 0), (1, 0), (0, 1), (1, 1))] == [13, 7, 5, 5]
b, c = counts[1, 0], counts[0, 1]
p = min(F(1), F(2 * sum(comb(b + c, k) for k in range(min(b, c) + 1)), 2 ** (b + c)))
means = [sum((pair[i] for pair in gaps), F()) / 30 for i in (0, 1)]
assert F(b - c, 30) == F(1, 15) and p == F(793, 1024)
assert means == [F(291331, 18849600), F(259, 33840)] and means[0] - means[1] == F(767993, 98436800)
print('PASS C1:', dict(counts), 'delta=', F(b-c, 30), 'p=', p, 'mean gaps=', means, 'gap difference=', means[0]-means[1])
```
